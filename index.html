// 0-indexed column positions in the sheet const G = { REG_ID: 0, // A SUBMISSION_ID: 1, // B EMAIL: 2, // C PHONE: 3, // D BRANCH: 4, // E NAME: 5, // F GENDER: 6, // G CATEGORY: 7, // H WORKFORCE: 8, // I LODGE: 9, // J — legacy, ignored RECEIPT_URL: 10, // K PAYMENT_REF: 11, // L PAYMENT_STATUS: 12, // M LATE_REG: 13, // N TICKET_SENT: 14, // O PERSON_NUM: 15, // P DAY1: 16, // Q — Day 1 check-in (Sep 3) DAY2: 17, // R — Day 2 check-in (Sep 4) DAY3: 18, // S — Day 3 check-in (Sep 5) QR_FILE_ID: 19, // T }; // 1-indexed for getRange/setValue const GCOL = { DAY1: 17, // Q DAY2: 18, // R DAY3: 19, // S }; function cors(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function doGet() { return cors({ status: 'Anapsyxis 2026 Gate API running.' }); } function doPost(e) { try { const body = JSON.parse(e.postData.contents); const action = body.action || ''; if (action === 'verify_pin') return cors(verifyPin(body)); if (action === 'scan') return cors(scanTicket(body)); if (action === 'checkin') return cors(checkIn(body)); if (action === 'stats') return cors(getStats(body)); if (action === 'manual') return cors(manualLookup(body)); return cors({ success: false, error: 'Unknown action' }); } catch (err) { return cors({ success: false, error: err.message }); } } // ── PIN ────────────────────────────────────────────────────── function verifyPin(body) { return (body.pin || '').toString().trim() === GATE_CONFIG.GATE_PIN ? { success: true } : { success: false, error: 'Incorrect PIN' }; } // ── Scan — look up by Reg ID, check per-day status ────────── function scanTicket(body) { const pin = (body.pin || '').toString().trim(); const regId = (body.regId || '').toString().trim().toUpperCase(); const day = parseInt(body.day) || 1; // 1, 2, or 3 if (pin !== GATE_CONFIG.GATE_PIN) return { success: false, error: 'Unauthorised' }; if (!regId) return { success: false, error: 'No Reg ID provided' }; if (day < 1 || day > 3) return { success: false, error: 'Invalid day' }; const sheet = SpreadsheetApp .openById(GATE_CONFIG.SPREADSHEET_ID) .getSheetByName(GATE_CONFIG.MASTER_TAB); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { const row = data[i]; if ((row[G.REG_ID] || '').toString().trim().toUpperCase() !== regId) continue; const payment = (row[G.PAYMENT_STATUS] || '').toString().trim().toLowerCase(); const dayCol = [G.DAY1, G.DAY2, G.DAY3][day - 1]; const checkedIn = (row[dayCol] || '').toString().trim(); const allDays = [ (row[G.DAY1] || '').toString().trim(), (row[G.DAY2] || '').toString().trim(), (row[G.DAY3] || '').toString().trim(), ]; if (payment !== 'confirmed') { return { success: false, status: 'UNCONFIRMED', message: 'Payment not confirmed. This ticket is not valid for entry.', attendee: buildAttendee(row, allDays), }; } if (checkedIn) { return { success: true, status: 'ALREADY_SCANNED', message: `Already checked in for ${GATE_CONFIG.DAY_LABELS[day - 1]} at ${checkedIn}`, attendee: buildAttendee(row, allDays), sheetRow: i + 1, day, }; } return { success: true, status: 'VALID', message: `Valid for ${GATE_CONFIG.DAY_LABELS[day - 1]}`, attendee: buildAttendee(row, allDays), sheetRow: i + 1, day, }; } return { success: false, status: 'NOT_FOUND', message: 'Ticket not found. Check QR or contact admin.', }; } // ── Check in — write timestamp to correct day column ───────── function checkIn(body) { const pin = (body.pin || '').toString().trim(); const sheetRow = parseInt(body.sheetRow); const day = parseInt(body.day) || 1; if (pin !== GATE_CONFIG.GATE_PIN) return { success: false, error: 'Unauthorised' }; if (!sheetRow || isNaN(sheetRow)) return { success: false, error: 'Invalid row' }; if (day < 1 || day > 3) return { success: false, error: 'Invalid day' }; const sheet = SpreadsheetApp .openById(GATE_CONFIG.SPREADSHEET_ID) .getSheetByName(GATE_CONFIG.MASTER_TAB); const checkInTime = Utilities.formatDate( new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm:ss' ); const dayColMap = { 1: GCOL.DAY1, 2: GCOL.DAY2, 3: GCOL.DAY3 }; sheet.getRange(sheetRow, dayColMap[day]).setValue(checkInTime); return { success: true, checkInTime, day }; } // ── Manual lookup by name, Reg ID, or branch ───────────────── function manualLookup(body) { const pin = (body.pin || '').toString().trim(); const query = (body.query || '').toString().trim().toLowerCase(); if (pin !== GATE_CONFIG.GATE_PIN) return { success: false, error: 'Unauthorised' }; if (query.length < 2) return { success: false, error: 'Query too short' }; const sheet = SpreadsheetApp .openById(GATE_CONFIG.SPREADSHEET_ID) .getSheetByName(GATE_CONFIG.MASTER_TAB); const data = sheet.getDataRange().getValues(); const results = []; for (let i = 1; i < data.length; i++) { const row = data[i]; const name = (row[G.NAME] || '').toLowerCase(); const regId = (row[G.REG_ID] || '').toLowerCase(); const branch = (row[G.BRANCH] || '').toLowerCase(); if (name.includes(query) || regId.includes(query) || branch.includes(query)) { const allDays = [ (row[G.DAY1] || '').toString().trim(), (row[G.DAY2] || '').toString().trim(), (row[G.DAY3] || '').toString().trim(), ]; results.push({ ...buildAttendee(row, allDays), sheetRow: i + 1 }); } if (results.length >= 10) break; } return { success: true, results }; } // ── Stats — total + per-day counts ─────────────────────────── function getStats(body) { const pin = (body.pin || '').toString().trim(); if (pin !== GATE_CONFIG.GATE_PIN) return { success: false, error: 'Unauthorised' }; const sheet = SpreadsheetApp .openById(GATE_CONFIG.SPREADSHEET_ID) .getSheetByName(GATE_CONFIG.MASTER_TAB); const data = sheet.getDataRange().getValues(); let total = 0, confirmed = 0, male = 0, female = 0; let day1 = 0, day2 = 0, day3 = 0; for (let i = 1; i < data.length; i++) { const row = data[i]; if (!(row[G.NAME] || '').toString().trim()) continue; total++; const payment = (row[G.PAYMENT_STATUS] || '').toString().toLowerCase(); const gender = (row[G.GENDER] || '').toString().trim(); if (payment === 'confirmed') confirmed++; if ((row[G.DAY1] || '').toString().trim()) day1++; if ((row[G.DAY2] || '').toString().trim()) day2++; if ((row[G.DAY3] || '').toString().trim()) day3++; if (gender === 'Male') male++; if (gender === 'Female') female++; } return { success: true, stats: { total, confirmed, male, female, day1, day2, day3 }, }; } // ── Helper: build attendee object ──────────────────────────── function buildAttendee(row, allDays) { return { regId: (row[G.REG_ID] || '').toString().trim(), name: (row[G.NAME] || '').toString().trim(), gender: (row[G.GENDER] || '').toString().trim(), category: (row[G.CATEGORY] || '').toString().trim(), branch: (row[G.BRANCH] || '').toString().trim(), workforce: (row[G.WORKFORCE] || '').toString().trim(), payment: (row[G.PAYMENT_STATUS] || '').toString().trim(), day1: allDays[0] || null, day2: allDays[1] || null, day3: allDays[2] || null, }; } // ── One-time setup: add day check-in column headers ────────── function setupGateColumns() { const sheet = SpreadsheetApp .openById(GATE_CONFIG.SPREADSHEET_ID) .getSheetByName(GATE_CONFIG.MASTER_TAB); const headers = { [GCOL.DAY1]: 'Day 1 Check-in (Sep 3)', [GCOL.DAY2]: 'Day 2 Check-in (Sep 4)', [GCOL.DAY3]: 'Day 3 Check-in (Sep 5)', }; for (const [col, label] of Object.entries(headers)) { const cell = sheet.getRange(1, parseInt(col)); if (!cell.getValue()) { cell.setValue(label) .setFontWeight('bold') .setBackground('#1B5E20') .setFontColor('#C9A227'); Logger.log(`✅ Added: ${label} at col ${col}`); } else { Logger.log(`Col ${col} already set: ${cell.getValue()}`); } } }
